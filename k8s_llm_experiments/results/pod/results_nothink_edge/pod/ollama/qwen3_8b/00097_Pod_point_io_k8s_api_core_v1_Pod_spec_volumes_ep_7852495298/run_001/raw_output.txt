```yaml
apiVersion: v1
kind: Pod
metadata:
  name: ephemeral-volume-pod
  labels:
    app: ephemeral-volume
spec:
  containers:
  - name: app-container
    image: nginx
    volumeMounts:
    - name: ephemeral-volume
      mountPath: /data
  volumes:
  - name: ephemeral-volume
    emptyDir:
      medium: Memory
      sizeLimit: 1Gi
```
