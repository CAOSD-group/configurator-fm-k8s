```yaml
apiVersion: v1
kind: Pod
metadata:
  name: ephemeral-container-pod
spec:
  containers:
  - name: main-container
    image: nginx
    resources:
      limits:
        memory: "128Mi"
        cpu: "500m"
  ephemeralContainers:
  - name: ephemeral-container
    image: busybox
    command: ["sh", "-c", "echo 'Ephemeral container running'; sleep 3600"]
    resources:
      limits:
        memory: "64Mi"
        cpu: "250m"
```
