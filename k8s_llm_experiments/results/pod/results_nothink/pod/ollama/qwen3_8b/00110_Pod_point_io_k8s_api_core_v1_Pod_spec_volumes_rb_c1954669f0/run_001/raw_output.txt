apiVersion: v1
kind: Pod
metadata:
  name: rbd-pod
spec:
  containers:
  - name: rbd-container

    command:
      - sleep
      - "3600"

    - name: rbd-volume
      mountPath: /mnt/rbd

  - name: rbd-volume
    persistentVolumeClaim:
      claimName: rbd-pvc
---

kind: PersistentVolumeClaim
metadata:
  name: rbd-pvc
spec:
  accessModes:
    - ReadWriteMany
  storageClassName: rbd
  resources:
    requests:
      storage: 1Gi
  volumeAttributes:
    rbdPool: myrbdpool
    rbdImage: myrbdimage
    rbdOptions: "size=1G,format=2"
    fsType: "ext4"
