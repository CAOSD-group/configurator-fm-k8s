```yaml
apiVersion: v1
kind: Pod
metadata:
  name: bind-mount-pod
spec:
  initContainers:
  - name: init-container
    image: busybox
    command:
    - sh
    - -c
    - echo "Init container running"
    volumeMounts:
    - name: test-volume
      mountPath: /mnt/test
      readOnly: true
      bindMountOptions:
        - ro
  containers:
  - name: main-container
    image: nginx
    volumeMounts:
    - name: test-volume
      mountPath: /usr/share/nginx/html
  volumes:
  - name: test-volume
    hostPath:
      path: /tmp/test
```
