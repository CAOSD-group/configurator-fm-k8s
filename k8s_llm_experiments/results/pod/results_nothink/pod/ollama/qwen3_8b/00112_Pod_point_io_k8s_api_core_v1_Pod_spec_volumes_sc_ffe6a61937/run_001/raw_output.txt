```yaml
apiVersion: v1
kind: Pod
metadata:
  name: scaleio-pod
spec:
  containers:
  - name: app-container
    image: nginx
    volumeMounts:
    - name: scaleio-volume
      mountPath: /data
  volumes:
  - name: scaleio-volume
    persistentVolumeClaim:
      claimName: scaleio-pvc
---
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: scaleio-pvc
spec:
  accessModes:
    - ReadWriteMany
  storageClassName: scaleio-storage-class
  resources:
    requests:
      storage: 1Gi
---
apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: scaleio-storage-class
provisioner: kubernetes.io/scaleio
parameters:
  storageMode: striped
```
