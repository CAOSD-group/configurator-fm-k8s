apiVersion: apps/v1
kind: Deployment
metadata:
  name: http-readiness-probe
spec:
  replicas: 1
  selector:
    matchLabels:
      app: http-readiness-probe
  template:
    metadata:
      labels:
        app: http-readiness-probe
    spec:
      containers:
      - name: http-container
        image: nginx:latest
        readinessProbe:
          httpGet:
            path: "/"
            port: 80
          initialDelaySeconds: 5
          periodSeconds: 10
          failureThreshold: 5
