---
apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: px-sc
provisioner: portworx.spc
parameters:
  fsType: ext4
---
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: px-pvc
spec:
  accessModes:
    - ReadWriteMany
  storageClassName: px-sc
  resources:
    requests:
      storage: 10Gi
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: px-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: px-app
  template:
    metadata:
      labels:
        app: px-app
    spec:
      containers:
      - name: px-container
        image: nginx
        volumeMounts:
        - name: px-volume
          mountPath: /data
      volumes:
      - name: px-volume
        persistentVolumeClaim:
          claimName: px-pvc
