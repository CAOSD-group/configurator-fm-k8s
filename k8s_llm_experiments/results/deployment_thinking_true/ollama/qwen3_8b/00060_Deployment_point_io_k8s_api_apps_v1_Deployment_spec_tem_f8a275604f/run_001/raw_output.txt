apiVersion: apps/v1
kind: Deployment
metadata:
  name: example-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: example
  template:
    metadata:
      labels:
        app: example
    spec:
      containers:
        - name: main-container
          image: nginx
          volumeMounts:
            - name: example-volume
              mountPath: /mnt
      volumes:
        - name: example-volume
          hostPath:
            path: /host/path
      ephemeralContainers:
        - name: ephemeral-container
          image: busybox
          volumeMounts:
            - name: example-volume
              mountPath: /mnt
              mountOptions:
                - bind
                - ro
