apiVersion: apps/v1
kind: Deployment
metadata:
  name: example-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: example
  template:
    metadata:
      labels:
        app: example
    spec:
      containers:
        - name: main
          image: nginx
      ephemeralContainers:
        - name: ephemeral
          image: busybox
          resources:
            limits:
              memory: "128Mi"
              cpu: "250m"
