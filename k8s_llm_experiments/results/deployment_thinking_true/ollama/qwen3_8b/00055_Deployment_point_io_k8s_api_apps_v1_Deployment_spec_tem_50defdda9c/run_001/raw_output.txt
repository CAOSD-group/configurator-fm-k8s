```
apiVersion: apps/v1
kind: Deployment
metadata:
  name: exit-code-demo
spec:
  replicas: 1
  selector:
    matchLabels:
      app: exit-code-demo
  template:
    metadata:
      labels:
        app: exit-code-demo
    spec:
      containers:
      - name: main
        image: nginx
        ports:
        - containerPort: 80
      ephemeralContainers:
      - name: exit-code-container
        image: alpine
        command:
        - sh
        - -c
        - echo "Testing exit code"; exit 0
        resources:
          limits:
            memory: "64Mi"
            cpu: "250m"
        restartPolicy: OnFailure
        exitCode: 0
```
