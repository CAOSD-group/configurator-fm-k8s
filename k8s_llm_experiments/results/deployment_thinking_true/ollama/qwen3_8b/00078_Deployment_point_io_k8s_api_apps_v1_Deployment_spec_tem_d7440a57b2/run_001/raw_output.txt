apiVersion: apps/v1
kind: Deployment
metadata:
  name: example-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: example
  template:
    metadata:
      labels:
        app: example
    spec:
      initContainers:
        - name: init-container
          image: busybox
          command: ["sh", "-c", "echo 'Init container running'; sleep 30"]
          ports:
            - containerPort: 8080
              name: http-port
          lifecycle:
            startupProbe:
              failureThreshold: 10
              httpGet:
                path: /health
                port: http-port
                scheme: HTTP
              initialDelaySeconds: 5
              periodSeconds: 5
      containers:
        - name: main-container
          image: nginx
          ports:
            - containerPort: 80
