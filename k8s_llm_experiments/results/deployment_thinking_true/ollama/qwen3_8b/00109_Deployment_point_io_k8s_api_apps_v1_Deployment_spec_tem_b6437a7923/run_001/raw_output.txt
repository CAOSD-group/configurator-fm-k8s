```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: iscsi-app
spec:
  replicas: 1
  selector:
    matchLabels:
      app: iscsi-app
  template:
    metadata:
      labels:
        app: iscsi-app
    spec:
      containers:
      - name: app
        image: nginx
        volumeMounts:
        - name: iscsi-volume
          mountPath: /data
      volumes:
      - name: iscsi-volume
        persistentVolumeClaim:
          claimName: iscsi-pvc
---
apiVersion: v1
kind: PersistentVolume
metadata:
  name: iscsi-pv
spec:
  capacity:
    storage: 10Gi
  accessModes:
    - ReadWriteOnce
  csi:
    driver: iscsi.csi.k8s.io
    volumeHandle: iqn.2003-01.com.example:storage.target1
    fsType: ext4
---
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: iscsi-pvc
spec:
  accessModes:
    - ReadWriteOnce
  resources:
    requests:
      storage: 10Gi
```
