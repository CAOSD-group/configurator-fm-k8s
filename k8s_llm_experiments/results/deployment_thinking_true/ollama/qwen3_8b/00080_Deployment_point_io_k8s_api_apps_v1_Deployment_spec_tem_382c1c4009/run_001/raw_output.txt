```
apiVersion: apps/v1
kind: Deployment
metadata:
  name: example-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: example
  template:
    metadata:
      labels:
        app: example
    spec:
      initContainers:
        - name: init-container
          image: busybox
          command: ["sh", "-c", "echo 'Init container running'"]
          volumeMounts:
            - name: my-volume
              mountPath: /mnt/data
              mountOptions:
                - ro
      containers:
        - name: main-container
          image: nginx
          volumeMounts:
            - name: my-volume
              mountPath: /usr/share/nginx/html
      volumes:
        - name: my-volume
          hostPath:
            path: /host/path
            type: Directory
```
