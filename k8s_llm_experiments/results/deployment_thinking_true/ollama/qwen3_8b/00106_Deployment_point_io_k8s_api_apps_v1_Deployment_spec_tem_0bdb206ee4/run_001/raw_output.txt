apiVersion: v1
kind: PersistentVolume
metadata:
  name: fc-pv
spec:
  capacity:
    storage: 10Gi
  accessModes:
    - ReadWriteOnce
  storageClassName: "fc-storage-class"
  csi:
    driver: com.example.fc-driver
    volumeHandle: "vol-1234"
    fsType: ext4
    lun: "1"
    targetiqn: "iqn.2001-08.com.example:target1"
---
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: fc-pvc
spec:
  accessModes:
    - ReadWriteOnce
  storageClassName: "fc-storage-class"
  resources:
    requests:
      storage: 10Gi
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: fc-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: fc-app
  template:
    metadata:
      labels:
        app: fc-app
    spec:
      containers:
        - name: fc-container
          image: nginx
          volumeMounts:
            - name: fc-volume
              mountPath: /usr/share/nginx/html
      volumes:
        - name: fc-volume
          persistentVolumeClaim:
            claimName: fc-pvc
