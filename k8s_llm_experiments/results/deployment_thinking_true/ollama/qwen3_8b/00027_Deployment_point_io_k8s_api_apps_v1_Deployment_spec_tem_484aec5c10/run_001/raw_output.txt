```
apiVersion: apps/v1
kind: Deployment
metadata:
  name: liveness-probe-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: liveness-probe
  template:
    metadata:
      labels:
        app: liveness-probe
    spec:
      containers:
      - name: liveness-container
        image: nginx:latest
        ports:
        - containerPort: 8080
        livenessProbe:
          tcpSocket:
            port: 8080
          initialDelaySeconds: 5
          periodSeconds: 10
```
