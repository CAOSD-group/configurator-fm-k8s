```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: probe-demo
spec:
  replicas: 1
  selector:
    matchLabels:
      app: probe-demo
  template:
    metadata:
      labels:
        app: probe-demo
    spec:
      initContainers:
        - name: init-container
          image: busybox
          command: ["sh", "-c", "sleep 3600"]
          startupProbe:
            tcpSocket:
              port: 8080
            initialDelaySeconds: 5
            periodSeconds: 10
            failureThreshold: 5
      containers:
        - name: main-container
          image: nginx
          ports:
            - containerPort: 80
```
