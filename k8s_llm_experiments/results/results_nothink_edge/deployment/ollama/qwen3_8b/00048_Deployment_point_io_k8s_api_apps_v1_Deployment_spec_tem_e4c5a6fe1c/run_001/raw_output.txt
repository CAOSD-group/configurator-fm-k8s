```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: liveness-probe-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: liveness-probe
  template:
    metadata:
      labels:
        app: liveness-probe
    spec:
      containers:
      - name: main-container
        image: nginx
        ports:
        - containerPort: 80
        livenessProbe:
          tcpSocket:
            port: 80
          initialDelaySeconds: 5
          periodSeconds: 10
      ephemeralContainers:
      - name: liveness-ephemeral
        image: nginx
        ports:
        - containerPort: 80
        livenessProbe:
          tcpSocket:
            port: 80
          initialDelaySeconds: 5
          periodSeconds: 10
```
