apiVersion: apps/v1
kind: Deployment
metadata:
  name: ephemeral-container-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: ephemeral-container
  template:
    metadata:
      labels:
        app: ephemeral-container
    spec:
      containers:
      - name: main-container
        image: nginx
        ports:
        - containerPort: 80
      - name: ephemeral-container
        image: busybox
        command: ["sh", "-c", "while true; do sleep 1; done"]
        volumeMounts:
        - name: test-volume
          mountPath: /data
          readOnly: true
          bindMountOptions:
            - ro
      volumes:
      - name: test-volume
        hostPath:
          path: /tmp/test
