apiVersion: apps/v1
kind: Deployment
metadata:
  name: example-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: example
  template:
    metadata:
      labels:
        app: example
    spec:
      affinity:
        podAffinity:
          - labelSelector:
              matchExpressions:

                  operator: In
                  values:
                    - example
              namespaceSelector:
                matchLabels:
                  app: example
            topologyKey: "kubernetes.io/hostname"
      containers:
        - name: example-container
          image: nginx:latest
