```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: storageos-app
spec:
  replicas: 1
  selector:
    matchLabels:
      app: storageos-app
  template:
    metadata:
      labels:
        app: storageos-app
    spec:
      containers:
      - name: app
        image: nginx
        volumeMounts:
        - name: storageos-volume
          mountPath: /data
          subPath: .
      volumes:
      - name: storageos-volume
        storageos:
          fsType: ext4
```
