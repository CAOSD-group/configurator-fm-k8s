```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: azure-disk-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: azure-disk-app
  template:
    metadata:
      labels:
        app: azure-disk-app
    spec:
      containers:
      - name: azure-disk-container
        image: nginx:latest
        volumeMounts:
        - name: azure-disk
          mountPath: /mnt/azure
      volumes:
      - name: azure-disk
        azureDisk:
          diskName: my-azure-disk
          diskURI: https://mystorageaccount.blob.core.windows.net/mycontainer/mydisk.vhd
          kind: AzureDisk
```
