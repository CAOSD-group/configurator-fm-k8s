```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: example-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: example
  template:
    metadata:
      labels:
        app: example
    spec:
      initContainers:
        - name: init-container
          image: busybox
          command:
            - sh
            - -c
            - echo "Init container running"
          resources:
            limits:
              memory: "128Mi"
              cpu: "250m"
            requests:
              memory: "64Mi"
              cpu: "125m"
          resizePolicy:
            - name: custom-resize-policy
              type: Memory
              min: 64Mi
              max: 128Mi
      containers:
        - name: main-container
          image: nginx
          ports:
            - containerPort: 80
```
