```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: example-deployment
  namespace: default
spec:
  replicas: 1
  selector:
    matchLabels:
      app: example
  template:
    metadata:
      labels:
        app: example
    spec:
      affinity:
        podAffinity:
          - labelSelector:
              matchExpressions:
                - key: app
                  operator: In
                  values:
                    - example
              namespaceSelector:
                matchExpressions:
                  - key: kubernetes.io/metadata.name
                    operator: In
                    values:
                      - my-namespace
            topologyKey: kubernetes.io/hostname
      containers:
        - name: example-container
          image: nginx:latest
          ports:
            - containerPort: 80
```
