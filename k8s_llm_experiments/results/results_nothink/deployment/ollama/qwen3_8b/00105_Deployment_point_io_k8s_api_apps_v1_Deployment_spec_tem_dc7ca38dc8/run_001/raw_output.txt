apiVersion: apps/v1

  name: example-deployment

    metadata:
      labels:
        app: example

      containers:

        image: nginx:latest
        ports:
        - containerPort: 80
      volumes:
      - name: example-volume
        persistentVolumeClaim:
          claimName: example-claim
      initContainers:
      - name: init-container
        image: busybox:latest
        command:
        - sh
        - -c
        - echo "Initializing..."
        volumeMounts:
        - name: example-volume
          mountPath: /data
      volumes:
      - name: example-volume
