```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: example-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: example
  template:
    metadata:
      labels:
        app: example
    spec:
      initContainers:
        - name: init-container
          image: busybox
          imagePullPolicy: IfNotPresent
          command:
            - sh
            - -c
            - echo "Init container running"
      containers:
        - name: main-container
          image: nginx
          ports:
            - containerPort: 80
```
