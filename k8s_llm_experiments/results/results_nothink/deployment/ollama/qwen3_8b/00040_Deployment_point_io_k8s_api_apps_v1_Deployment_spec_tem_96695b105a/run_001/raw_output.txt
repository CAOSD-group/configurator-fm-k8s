```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: dns-policy-demo
spec:
  replicas: 1
  selector:
    matchLabels:
      app: dns-policy-demo
  template:
    metadata:
      labels:
        app: dns-policy-demo
    spec:
      dnsPolicy: ClusterFirstWithHostNet
      containers:
      - name: alpine
        image: alpine:3.18
        command:
        - sh
        - -c
        - echo "DNS Policy: $(cat /etc/resolv.conf | grep nameserver | awk '{print $2}')"
```
